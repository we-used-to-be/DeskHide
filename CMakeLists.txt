cmake_minimum_required(VERSION 3.21)
project(DeskHide VERSION 1.0.0 LANGUAGES CXX RC)
# CMake can decode the Chinese-only MSVC probe as UTF-8 while the host uses
# code page 936. build.ps1 runs the compiler under 65001; repair that cache text.
if(CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^娉ㄦ剰")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC on Windows (scripts/build.ps1).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
string(REPLACE "/O2" "/O1" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- $<$<CONFIG:Release>:/GL>)
add_link_options($<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
add_library(desktop_core STATIC src/desktop.cpp src/startup.cpp)
target_include_directories(desktop_core PUBLIC src)
target_link_libraries(desktop_core PUBLIC user32 shell32 ole32 oleaut32 oleacc advapi32 uuid)
add_executable(DeskHide WIN32 src/main.cpp assets/app.rc)
target_link_libraries(DeskHide PRIVATE desktop_core)
target_link_options(DeskHide PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/assets/app.manifest")
enable_testing()
add_executable(click_tests tests/click_tests.cpp)
target_include_directories(click_tests PRIVATE src)
add_test(NAME click_detector COMMAND click_tests)
add_executable(windows_tests tests/windows_tests.cpp)
target_link_libraries(windows_tests PRIVATE desktop_core)
add_test(NAME startup_registry COMMAND windows_tests --registry)
